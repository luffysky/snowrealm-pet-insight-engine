// 新增：src/app/api/quiz-result/[responseId]/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { queryOne } from '@/lib/db'
import { buildStoryCard } from '@/lib/storyCardEngine'

interface ResponseInfo extends Record<string, unknown> {
  id: string
  survey_id: string
  survey_title: string
  survey_slug: string
  tenant_id: string
  personality_result: string | null  // JSON: { title, emoji, description }
}

export async function GET(
  _req: NextRequest,
  { params }: { params: Promise<{ responseId: string }> }
) {
  const { responseId } = await params

  const response = await queryOne<ResponseInfo>(
    `SELECT r.id, r.survey_id,
            s.title AS survey_title, s.slug AS survey_slug, s.tenant_id,
            r.personality_result
     FROM responses r
     JOIN surveys s ON s.id = r.survey_id
     WHERE r.id = $1 AND r.status = 'completed'`,
    [responseId]
  )

  if (!response) {
    return NextResponse.json({ error: 'not_found' }, { status: 404 })
  }

  // 人格結果
  let title = '感謝你完成測驗！'
  let emoji = '✨'
  let description = ''

  if (response.personality_result) {
    try {
      const parsed = JSON.parse(response.personality_result as string)
      title = parsed.title || title
      emoji = parsed.emoji || emoji
      description = parsed.description || description
    } catch { /* ignore */ }
  }

  // Story Card
  const storyCard = await buildStoryCard(responseId)

  return NextResponse.json({
    title,
    emoji,
    description,
    storyCard,
    surveyTitle: response.survey_title,
    surveySlug: response.survey_slug,
    tenantSlug: response.tenant_id,
  })
}

// 新增：src/lib/quizResultEngine.ts

/**
 * 測驗結果計算引擎
 * 根據填答的選項分數加總，對應到預設的人格/角色結果
 */

export interface QuizResultType {
  id: string
  title: string        // 「熱血系全能冒險者」
  emoji: string        // ⭐
  description: string  // 一段 2-3 行的描述
  minScore: number     // 分數區間下限
  maxScore: number     // 分數區間上限
}

export interface OptionWeight {
  questionKey: string
  optionValue: string
  score: number
}

export interface QuizConfig {
  results: QuizResultType[]
  weights: OptionWeight[]
}

/**
 * 根據填答計算人格結果
 * @param answers Map<questionKey, answer value or values>
 * @param config 測驗的結果配置
 */
export function calculateQuizResult(
  answers: Map<string, string | string[]>,
  config: QuizConfig
): QuizResultType {
  let totalScore = 0

  for (const [key, value] of answers) {
    const values = Array.isArray(value) ? value : [value]
    for (const v of values) {
      const weight = config.weights.find(
        w => w.questionKey === key && w.optionValue === v
      )
      if (weight) totalScore += weight.score
    }
  }

  // 找到對應的結果（分數區間）
  const sorted = [...config.results].sort((a, b) => a.minScore - b.minScore)
  const matched = sorted.find(r => totalScore >= r.minScore && totalScore <= r.maxScore)

  // fallback 到最後一個
  return matched || sorted[sorted.length - 1] || {
    id: 'default',
    title: '神秘旅人',
    emoji: '🌟',
    description: '你是一個獨特的存在，無法被簡單分類。',
    minScore: 0,
    maxScore: 999,
  }
}

import { NextRequest, NextResponse } from 'next/server'
import { getOrCreateReferralCode, getReferralByCode } from '@/lib/referralEngine'
import { queryOne } from '@/lib/db'

// GET /api/referral?code=XXX — 查詢邀請碼資訊（公開，invite 頁面用）
export async function GET(req: NextRequest) {
  const code = req.nextUrl.searchParams.get('code')
  if (!code) return NextResponse.json({ valid: false })

  const referral = await getReferralByCode(code)
  if (!referral) return NextResponse.json({ valid: false })

  // 查推薦人名稱
  const user = await queryOne<{ display_name: string }>(
    'SELECT display_name FROM admin_users WHERE id = $1',
    [referral.referrer_id]
  )

  return NextResponse.json({
    valid: true,
    displayName: user?.display_name || '朋友',
  })
}

// POST /api/referral — 生成我的邀請碼（需登入）
export async function POST(req: NextRequest) {
  const userId = req.headers.get('x-user-id')
  if (!userId) return NextResponse.json({ error: '未登入' }, { status: 401 })

  const code = await getOrCreateReferralCode(userId)
  const baseUrl = process.env.NEXT_PUBLIC_BASE_URL || 'https://insight.snowrealm.pet'

  return NextResponse.json({
    code,
    url: `${baseUrl}/invite/${code}`,
  })
}
